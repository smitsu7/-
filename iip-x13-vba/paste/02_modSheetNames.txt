Private Function IIPU(ByVal hexText As String) As String
    Static cache As Object
    Dim i As Long, result As String
    If cache Is Nothing Then Set cache = CreateObject("Scripting.Dictionary")
    If cache.Exists(hexText) Then
        IIPU = cache(hexText)
        Exit Function
    End If
    For i = 1 To Len(hexText) Step 4
        result = result & ChrW$(CLng("&H" & Mid$(hexText, i, 4)))
    Next i
    cache.Add hexText, result
    IIPU = result
End Function
