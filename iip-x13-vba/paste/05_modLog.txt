Public Sub WriteLog(ByVal procedureName As String, ByVal levelName As String, ByVal messageText As String)
    Dim ws As Worksheet
    Dim nextRow As Long

    On Error GoTo Fallback
    Set ws = GetLogSheet()
    nextRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row + 1
    If nextRow < 2 Then nextRow = 2
    ws.Cells(nextRow, 1).Value = Now
    ws.Cells(nextRow, 1).NumberFormat = "yyyy-mm-dd hh:mm:ss"
    ws.Cells(nextRow, 2).NumberFormat = "@"
    ws.Cells(nextRow, 2).Resize(1, 5).Value2 = Array(OneLine(procedureName), UCase$(OneLine(levelName)), _
        OneLine(messageText), Application.UserName, ThisWorkbook.Name)
    Exit Sub
Fallback:
    Debug.Print Format$(Now, "yyyy-mm-dd hh:nn:ss"), procedureName, levelName, messageText
End Sub

Private Function GetLogSheet() As Worksheet
    Dim ws As Worksheet
    Set ws = modMain.WorksheetOrNothing(modSheetNames.LogSheet())
    If ws Is Nothing Then
        Set ws = modMain.EnsureWorksheet(modSheetNames.LogSheet())
        ws.Cells(1, 1).Resize(1, 6).Value2 = Array((IIPU("65E56642")), (IIPU("51E67406540D")), (IIPU("91CD89815EA6")), (IIPU("51855BB9")), (IIPU("0045007800630065006C522975288005")), (IIPU("30D630C330AF")))
        ws.Rows(1).Font.Bold = True
        ws.Columns("D").ColumnWidth = 90
    End If
    Set GetLogSheet = ws
End Function

Private Function OneLine(ByVal text As String) As String
    OneLine = Replace(Replace(text, vbCr, " "), vbLf, " ")
End Function

Private Function IIPU(ByVal hexText As String) As String
    Static cache As Object
    Dim i As Long, result As String
    If cache Is Nothing Then Set cache = CreateObject("Scripting.Dictionary")
    If cache.Exists(hexText) Then
        IIPU = cache(hexText)
        Exit Function
    End If
    For i = 1 To Len(hexText) Step 4
        result = result & ChrW$(CLng("&H" & Mid$(hexText, i, 4)))
    Next i
    cache.Add hexText, result
    IIPU = result
End Function
