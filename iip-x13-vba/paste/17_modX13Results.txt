    For Each line In lines
        pos = InStr(CStr(line), ":")

            key = Trim$(Left$(CStr(line), pos - 1))

            End If
        End If
    Next line

Private Function InverseMatrix(ByRef matrix() As Double, ByVal k As Long, ByRef reason As String) As Variant
    Dim a() As Double, i As Long, j As Long, r As Long, pivot As Long, factor As Double, temp As Double, best As Double

        For j = 1 To k

        Next j
        a(i, k + i) = 1#
    Next i
    For j = 1 To k
        pivot = j
        best = Abs(a(j, j))
        For r = j + 1 To k
            If Abs(a(r, j)) > best Then best = Abs(a(r, j)): pivot = r
        Next r
        If best < 1E-300 Then reason = (IIPU("517152066563884C5217304C72797570")): Exit Function

            For i = 1 To 2 * k
                temp = a(j, i): a(j, i) = a(pivot, i): a(pivot, i) = temp
            Next i
        End If
        factor = a(j, j)
        For i = 1 To 2 * k
            a(j, i) = a(j, i) / factor
        Next i
        For r = 1 To k
            If r <> j Then
                factor = a(r, j)
                For i = 1 To 2 * k
                    a(r, i) = a(r, i) - factor * a(j, i)
                Next i
            End If
        Next r
    Next j
