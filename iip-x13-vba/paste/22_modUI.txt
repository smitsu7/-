Public Function DurationText(ByVal seconds As Double) As String
    Dim whole As Long
    If seconds < 60 Then
        DurationText = modX13IO.NumberText(Round(seconds, 1)) & (IIPU("79D2"))
        Exit Function
    End If
    whole = CLng(Int(seconds))
    If whole >= 3600 Then
        DurationText = (whole \ 3600) & (IIPU("66429593")) & Format$((whole Mod 3600) \ 60, "00") & (IIPU("5206"))

    End If

Private Function AsciiText(ByVal text As String) As String
    Dim i As Long, code As Long
    For i = 1 To Len(text)
        code = AscW(Mid$(text, i, 1))
        If code = 9 Then
            Mid$(text, i, 1) = " "
        ElseIf code < 32 Or code > 126 Then
            Mid$(text, i, 1) = "?"
        End If
    Next i
    AsciiText = text
End Function

Private Function IIPU(ByVal hexText As String) As String
    Static cache As Object
    Dim i As Long, result As String
    If cache Is Nothing Then Set cache = CreateObject("Scripting.Dictionary")
    If cache.Exists(hexText) Then
        IIPU = cache(hexText)
        Exit Function
    End If
    For i = 1 To Len(hexText) Step 4
        result = result & ChrW$(CLng("&H" & Mid$(hexText, i, 4)))
    Next i
    cache.Add hexText, result
    IIPU = result
End Function
